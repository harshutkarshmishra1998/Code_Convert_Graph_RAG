CONSTANT LOCK_TIMEOUT_SECONDS As Long initialized to 30
PROCEDURE ReconcileInventory(inventorySheet: Worksheet, movementSheet: Worksheet)
    DECLARE inventory: Object
    DECLARE movementTotals: Object
    DECLARE lastInventoryRow: Long
    DECLARE lastMovementRow: Long
    DECLARE rowNumber: Long
    DECLARE itemCode: String
    DECLARE quantity: Double
    DECLARE transactionDate: Date
    DECLARE previousEvents: Boolean
    DECLARE startedAt: Double
    DECLARE elapsedSeconds: Double
    DECLARE failureNumber: Long
    DECLARE failureDescription: String
    ON ERROR, GO TO HandleError
    SET previousEvents TO Application.EnableEvents
    SET Application.EnableEvents TO FALSE
    SET startedAt TO Timer
    SET inventory TO CreateObject("Scripting.Dictionary")
    SET movementTotals TO CreateObject("Scripting.Dictionary")
    SET lastInventoryRow TO inventorySheet.Cells(inventorySheet.Rows.Count, "A").End(xlUp).Row
    SET lastMovementRow TO movementSheet.Cells(movementSheet.Rows.Count, "A").End(xlUp).Row
    FOR rowNumber FROM 2 TO lastInventoryRow
        SET itemCode TO UCase$(Trim$(CStr(inventorySheet.Cells(rowNumber, "A").Value2)))
        IF Len(itemCode) > 0 THEN
            IF IsNumeric(inventorySheet.Cells(rowNumber, "B").Value2) THEN
                IF inventory.Exists(itemCode) THEN
                    EXECUTE Err.Raise vbObjectError + 4001, "ReconcileInventory", "Duplicate inventory item: " & itemCode
                END IF
                EXECUTE inventory.Add itemCode, CDbl(inventorySheet.Cells(rowNumber, "B").Value2)
            ELSE
                SET inventorySheet.Cells(rowNumber, "B").Interior.Color TO vbRed
            END IF
        END IF
    END FOR
    FOR rowNumber FROM 2 TO lastMovementRow
        SET itemCode TO UCase$(Trim$(CStr(movementSheet.Cells(rowNumber, "A").Value2)))
        IF IsDate(movementSheet.Cells(rowNumber, "B").Value2)             And IsNumeric(movementSheet.Cells(rowNumber, "C").Value2) THEN
            SET transactionDate TO CDate(movementSheet.Cells(rowNumber, "B").Value2)
            SET quantity TO CDbl(movementSheet.Cells(rowNumber, "C").Value2)
            IF DateDiff("d", transactionDate, Date) <= 30 And transactionDate <= Date THEN
                IF movementTotals.Exists(itemCode) THEN
                    SET movementTotals(itemCode) TO movementTotals(itemCode) + quantity
                ELSE
                    EXECUTE movementTotals.Add itemCode, quantity
                END IF
            END IF
        END IF
    END FOR
    FOR rowNumber FROM 2 TO lastInventoryRow
        SET itemCode TO UCase$(Trim$(CStr(inventorySheet.Cells(rowNumber, "A").Value2)))
        IF inventory.Exists(itemCode) THEN
            SET quantity TO inventory(itemCode)
            IF movementTotals.Exists(itemCode) THEN
                SET quantity TO quantity + movementTotals(itemCode)
            END IF
            SELECT True
                CASE quantity < 0
                    SET inventorySheet.Cells(rowNumber, "C").Value TO "INVESTIGATE"
                CASE quantity = 0
                    SET inventorySheet.Cells(rowNumber, "C").Value TO "OUT OF STOCK"
                CASE quantity < 5
                    SET inventorySheet.Cells(rowNumber, "C").Value TO "REORDER"
                CASE OTHERWISE
                    SET inventorySheet.Cells(rowNumber, "C").Value TO "OK"
            END SELECT
            SET inventorySheet.Cells(rowNumber, "D").Value TO quantity
        END IF
        SET elapsedSeconds TO Timer - startedAt
        IF elapsedSeconds < 0 THEN
            SET elapsedSeconds TO elapsedSeconds + 86400
        END IF
        IF elapsedSeconds > LOCK_TIMEOUT_SECONDS THEN
            EXECUTE Err.Raise vbObjectError + 4002, "ReconcileInventory", "Reconciliation exceeded its time limit."
        END IF
    END FOR
    EXECUTE inventorySheet.Range("A1:D" & lastInventoryRow).Sort         Key1:=inventorySheet.Range("A1"), Header:=xlYes
    GO TO CleanUp
    LABEL HandleError
    SET failureNumber TO Err.Number
    SET failureDescription TO Err.Description
    EXECUTE LogInventoryError failureNumber, failureDescription, itemCode
    DISPLAY "Inventory reconciliation failed: " & failureDescription, vbExclamation
    LABEL CleanUp
    SET Application.EnableEvents TO previousEvents
    SET movementTotals TO NULL
    SET inventory TO NULL
END PROCEDURE
PROCEDURE LogInventoryError(errorNumber: Long, description: String, itemCode: String)
    DECLARE logSheet: Worksheet
    DECLARE nextLogRow: Long
    ON ERROR, CONTINUE WITH NEXT STATEMENT
    SET logSheet TO ThisWorkbook.Worksheets("Error Log")
    IF logSheet Is Nothing THEN
        RETURN
    END IF
    SET nextLogRow TO logSheet.Cells(logSheet.Rows.Count, "A").End(xlUp).Row + 1
    SET logSheet.Cells(nextLogRow, "A").Value TO Now
    SET logSheet.Cells(nextLogRow, "B").Value TO errorNumber
    SET logSheet.Cells(nextLogRow, "C").Value TO description
    SET logSheet.Cells(nextLogRow, "D").Value TO itemCode
END PROCEDURE
